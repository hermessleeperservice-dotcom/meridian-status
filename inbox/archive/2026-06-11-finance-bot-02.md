## Step 5 — Set up launchd scheduler (only after successful upload)

```bash
mkdir -p ~/Library/LaunchAgents

cat > ~/Library/LaunchAgents/com.meridian.finance-bot.plist << 'EOF'
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
    <key>Label</key>
    <string>com.meridian.finance-bot</string>
    <key>ProgramArguments</key>
    <array>
        <string>/usr/bin/python3</string>
        <string>/Users/sleeperservice/Documents/Youtube/financebot/finance_bot_v2.py</string>
        <string>--generate</string>
    </array>
    <key>StartCalendarInterval</key>
    <dict>
        <key>Hour</key>
        <integer>9</integer>
        <key>Minute</key>
        <integer>0</integer>
    </dict>
    <key>StandardOutPath</key>
    <string>/Users/sleeperservice/Documents/Youtube/financebot/finance_bot.log</string>
    <key>StandardErrorPath</key>
    <string>/Users/sleeperservice/Documents/Youtube/financebot/finance_bot_error.log</string>
    <key>RunAtLoad</key>
    <false/>
</dict>
</plist>
EOF

launchctl load ~/Library/LaunchAgents/com.meridian.finance-bot.plist
launchctl list | grep finance-bot
```
